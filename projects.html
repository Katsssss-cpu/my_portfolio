public void InitializeGame()
{
    Console.WriteLine("Starting the application...");
    int actionChoice;

    DisplayStatus(companionName, companionType, health, happiness, hunger, energy);

    do
    {
        DisplayActionMenu(currentLocation);

        string userInput = Console.ReadLine();
        try
        {
            actionChoice = Convert.ToInt32(userInput);
        }
        catch (Exception)
        {
            Console.WriteLine("Enter a whole number from the menu.");
            actionChoice = 0;
            continue;
        }

        switch (actionChoice)
        {
            case 1:
                if (companionType == "cat" || companionType == "dog")
                {
                    health = ChangeStat(health, 15);
                    hunger = ChangeStat(hunger, -10);
                    Console.WriteLine();
                    Console.WriteLine($"{companionName} eats the apple.");
                }
                else if (companionType == "toaster")
                {
                    Console.WriteLine();
                    Console.WriteLine("The toaster refuses the apple. Statistics unchanged.");
                }

                DisplayStatus(companionName, companionType, health, happiness, hunger, energy);

                Console.WriteLine();
                Console.WriteLine("Press any key to continue...");
                Console.ReadKey();
                break;
        }
    } while (actionChoice != 0);
}
